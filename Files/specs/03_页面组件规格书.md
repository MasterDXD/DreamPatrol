## 一、组件树

```
App
└── Layout
    ├── Navbar (标题"巡梦" | 日历入口 | 新建按钮)
    └── <Outlet /> (路由内容区)
        ├── HomePage (/)
        │   ├── DreamCard[] (梦境卡片列表)
        │   ├── EmptyState (无梦境时)
        │   └── FAB (新建梦境浮动按钮)
        ├── NewDreamPage (/new)
        │   ├── Textarea (文本输入)
        │   ├── VoiceInput (语音按钮)
        │   ├── EmotionPicker (情绪选择器)
        │   └── SubmitButton (提交)
        ├── DreamDetailPage (/dream/:id)
        │   ├── 情绪氛围背景
        │   ├── 梦境内容展示
        │   ├── 情绪标签
        │   ├── 日期时间
        │   └── 操作按钮 (编辑/删除/返回)
        ├── CalendarPage (/calendar)
        │   ├── Calendar (月视图日历)
        │   │   ├── 月份切换 (← →)
        │   │   └── 日期单元格 (含情绪标记点)
        │   └── DreamList (选中日期的梦境列表)
        └── EditDreamPage (/dream/:id/edit)
            └── (复用NewDreamPage组件结构)

全局浮层组件：
├── Toast (操作反馈提示)
└── ConfirmDialog (删除确认弹窗)
```

## 四、NewDreamPage（新建梦境）

| 属性 | 值 |
|------|-----|
| 文件路径 | `src/pages/NewDreamPage/NewDreamPage.tsx` |
| 路由 | `/new` |
| Props | 无 |

### 4.1 内部State

```typescript
const [content, setContent] = useState<string>('');
const [emotion, setEmotion] = useState<EmotionType>('calm'); // 默认"平静"
const [isSubmitting, setIsSubmitting] = useState<boolean>(false);
const [toast, setToast] = useState<{ message: string; type: 'success' | 'error' } | null>(null);
```

### 4.2 子组件使用

| 子组件 | Props映射 |
|--------|----------|
| `VoiceInput` | `onTranscript={(text) => setContent(prev => prev + text)}` |
| `EmotionPicker` | `value={emotion}` `onChange={setEmotion}` |
| `Toast` | 按toast状态控制 |

### 4.3 提交流程

```
1. 用户点击"留下这个梦"按钮
2. 校验 content.trim() 是否为空
   ├─ 为空 → Toast({ message: "梦还空着呢，写点什么吧", type: "error" })
   └─ 不为空 → 继续
3. setIsSubmitting(true)
4. 调用 createDream({ content: content.trim(), emotion })
5. dispatch({ type: 'ADD_DREAM', payload: newDream })
6. setIsSubmitting(false)
7. 显示成功Toast → navigate('/') 跳转回首页
```

### 4.4 渲染结构

```tsx
<div className={styles.page}>
  <h2 className={styles.heading}>记录你的梦</h2>
  
  {/* 文本输入区 */}
  <textarea
    className={styles.textarea}
    value={content}
    onChange={e => setContent(e.target.value)}
    placeholder="昨晚，你梦见了什么…"
    autoFocus
    rows={8}
    maxLength={5000}
  />
  <span className={styles.charCount}>
    {content.length}/5000
    {content.length > 4500 && <span className={styles.warning}> 快要写满了呢</span>}
  </span>

  {/* 语音录入 */}
  <VoiceInput onTranscript={(text) => setContent(prev => prev + ' ' + text)} />

  {/* 情绪选择 */}
  <div className={styles.section}>
    <label className={styles.label}>这个梦的情绪是…</label>
    <EmotionPicker value={emotion} onChange={setEmotion} />
  </div>

  {/* 提交按钮 */}
  <button
    className={styles.submitBtn}
    onClick={handleSubmit}
    disabled={isSubmitting}
  >
    {isSubmitting ? '正在安放这个梦...' : '留下这个梦'}
  </button>

  {/* Toast */}
  {toast && <Toast message={toast.message} type={toast.type} visible={!!toast} onClose={() => setToast(null)} />}
</div>
```

## 七、DreamDetailPage（梦境详情）

| 属性 | 值 |
|------|-----|
| 文件路径 | `src/pages/DreamDetailPage/DreamDetailPage.tsx` |
| 路由 | `/dream/:id` |
| Props | 无（从URL参数获取id） |

### 7.1 内部State与逻辑

```typescript
const { id } = useParams<{ id: string }>();
const { dreams } = useDreamContext();
const [showDeleteConfirm, setShowDeleteConfirm] = useState(false);

const dream = dreams.find(d => d.id === id);
const emotionMeta = dream ? EMOTION_META[dream.emotion] : null;
```

### 7.2 渲染分支

```tsx
// 分支1：dream不存在 → <EmptyState message="找不到这个梦" actionLabel="返回首页" onAction={() => navigate('/')} />
// 分支2：dream存在 → 正常渲染

<div 
  className={styles.page}
  style={{ background: emotionMeta?.bgGradient }}
>
  <button className={styles.backBtn} onClick={() => navigate(-1)}>← 返回</button>
  
  <div className={styles.content}>
    <div className={styles.emotionBadge}>
      {emotionMeta?.icon} {emotionMeta?.label}
    </div>
    
    <h2 className={styles.title}>{dream.title}</h2>
    
    <div className={styles.date}>
      {formatDate(dream.recordedDate)} {formatTime(dream.createdAt)}
    </div>
    
    <div className={styles.body}>
      {dream.content.split('\n').map((line, i) => (
        <p key={i}>{line}</p>
      ))}
    </div>
    
    <div className={styles.actions}>
      <button onClick={() => navigate(`/dream/${dream.id}/edit`)}>编辑</button>
      <button className={styles.dangerBtn} onClick={() => setShowDeleteConfirm(true)}>删除</button>
    </div>
  </div>
  
  {showDeleteConfirm && (
    <ConfirmDialog
      title="删除梦境"
      message="删除后，这个梦就真的消失了"
      confirmLabel="确认删除"
      onConfirm={handleDelete}
      onCancel={() => setShowDeleteConfirm(false)}
      visible={showDeleteConfirm}
    />
  )}
</div>
```

### 7.3 删除逻辑

```typescript
const handleDelete = () => {
  if (!dream) return;
  deleteDream(dream.id);
  dispatch({ type: 'DELETE_DREAM', payload: dream.id });
  setShowDeleteConfirm(false);
  navigate('/');
};
```

**CSS关键样式：**
- `page`: min-height 100vh，背景使用情绪bgGradient，transition background 500ms
- `content`: 最大宽度680px，margin auto，padding 32px，背景白色半透明，border-radius 16px
- `body`: line-height 1.8，font-size 16px
