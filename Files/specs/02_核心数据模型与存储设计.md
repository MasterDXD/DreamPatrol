## 一、核心数据类型

### 1.1 情绪枚举

```typescript
// src/types/dream.ts

/** 情绪类型枚举 */
export type EmotionType = 'joy' | 'calm' | 'sadness' | 'fear' | 'wonder' | 'nostalgia';
```

### 1.2 梦境主数据结构

```typescript
/** 单条梦境记录 */
export interface Dream {
  /** 唯一标识，使用 crypto.randomUUID() 生成 */
  id: string;

  /** 用户标识，用于数据隔离 */
  userId: string;

  /** 梦境标题，自动取内容前30个字符或用户自定义 */
  title: string;

  /** 梦境正文内容（纯文本，不含HTML） */
  content: string;

  /** 主导情绪 */
  emotion: EmotionType;

  /** 创建时间，ISO 8601格式 */
  createdAt: string;

  /** 最后修改时间，ISO 8601格式 */
  updatedAt: string;

  /** 梦境发生日期（YYYY-MM-DD），默认为创建日期 */
  recordedDate: string;
}
```

### 1.3 新建梦境输入

```typescript
/** 新建/编辑梦境的表单数据 */
export interface DreamInput {
  content: string;
  emotion: EmotionType;
  /** 可选，编辑时才传 */
  id?: string;
}
```

### 1.4 情绪元数据

```typescript
/** 单条情绪元数据 */
export interface EmotionMeta {
  value: EmotionType;
  label: string;
  icon: string;   // emoji字符
  color: string;   // HEX色值
  bgGradient: string;  // CSS gradient字符串，用于背景
}

/** 所有情绪元数据映射表 */
export type EmotionMetaMap = Record<EmotionType, EmotionMeta>;
```

### 1.5 全局应用状态

```typescript
/** 全局应用状态 */
export interface AppState {
  /** 所有梦境列表 */
  dreams: Dream[];
  /** 数据是否正在加载 */
  loading: boolean;
  /** 错误信息，null表示无错误 */
  error: string | null;
  /** 当前用户ID */
  userId: string;
}

/** 状态变更动作 */
export type AppAction =
  | { type: 'SET_DREAMS'; payload: Dream[] }
  | { type: 'ADD_DREAM'; payload: Dream }
  | { type: 'UPDATE_DREAM'; payload: Dream }
  | { type: 'DELETE_DREAM'; payload: string }  // payload = dream.id
  | { type: 'SET_LOADING'; payload: boolean }
  | { type: 'SET_ERROR'; payload: string | null };
```

## 三、存储层API（services/storage.ts 签名）

> 以下为 `src/services/storage.ts` 中必须实现的全部导出函数签名。
> AI应严格按此签名实现，不得增删参数。

```typescript
// src/services/storage.ts

// ===== 常量 =====
/** localStorage key前缀 */
export const STORAGE_PREFIX = 'dreampatrol';

// ===== 用户标识 =====

/**
 * 获取或创建当前用户ID
 * - 如果localStorage中已存在userId，直接返回
 * - 否则使用crypto.randomUUID()生成新ID并存储
 */
export function getUserId(): string;

// ===== 梦境CRUD =====

/**
 * 加载当前用户的所有梦境
 * @returns 梦境数组，按createdAt倒序排列；无数据时返回空数组[]
 */
export function loadDreams(): Dream[];

/**
 * 保存新梦境
 * @param input - 梦境内容与情绪
 * @returns 创建完成的Dream对象（含自动生成的id、userId、时间戳）
 */
export function createDream(input: DreamInput): Dream;

/**
 * 更新已有梦境
 * @param id - 要更新的梦境ID
 * @param input - 新的内容和情绪
 * @returns 更新后的Dream对象；如果id不存在则抛出Error
 */
export function updateDream(id: string, input: DreamInput): Dream;

/**
 * 删除梦境
 * @param id - 要删除的梦境ID
 * @returns void；如果id不存在则静默忽略
 */
export function deleteDream(id: string): void;

/**
 * 根据ID获取单条梦境
 * @param id - 梦境ID
 * @returns Dream对象；如果不存在返回undefined
 */
export function getDreamById(id: string): Dream | undefined;

/**
 * 根据日期获取梦境列表
 * @param dateStr - 日期字符串 YYYY-MM-DD
 * @returns 该日期的梦境数组
 */
export function getDreamsByDate(dateStr: string): Dream[];

/**
 * 获取所有有梦境的日期集合
 * @returns 日期字符串数组 YYYY-MM-DD[]
 */
export function getRecordedDates(): string[];
```

## 五、数据校验规则

| 字段 | 规则 | 违反时的处理 |
|------|------|-------------|
| `content` | 必填，不可为空或纯空白字符串 | 阻止提交，Toast"梦还空着呢，写点什么吧" |
| `content` | 最大长度5000字符 | 允许提交但Toast"这个梦很长呢，会被完整保存" |
| `emotion` | 必须是EmotionType枚举值之一 | 提交前默认选中第一个情绪 |
| `title` | 自动生成：取content前30字符（去除首尾空白） | 存储时自动处理 |
