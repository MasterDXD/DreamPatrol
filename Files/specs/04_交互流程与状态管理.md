## 一、全局状态管理

### 1.1 DreamContext

```typescript
// src/contexts/DreamContext.tsx

import React, { createContext, useContext, useReducer, useEffect, ReactNode } from 'react';
import { AppState, AppAction, Dream } from '../types/dream';
import { loadDreams, getUserId } from '../services/storage';

// ===== 初始状态 =====
const initialState: AppState = {
  dreams: [],
  loading: true,
  error: null,
  userId: '',
};

// ===== Reducer =====
function dreamReducer(state: AppState, action: AppAction): AppState {
  switch (action.type) {
    case 'SET_DREAMS':
      return { ...state, dreams: action.payload, loading: false };
    
    case 'ADD_DREAM':
      return { ...state, dreams: [action.payload, ...state.dreams] };
    
    case 'UPDATE_DREAM':
      return {
        ...state,
        dreams: state.dreams.map(d => d.id === action.payload.id ? action.payload : d),
      };
    
    case 'DELETE_DREAM':
      return {
        ...state,
        dreams: state.dreams.filter(d => d.id !== action.payload),
      };
    
    case 'SET_LOADING':
      return { ...state, loading: action.payload };
    
    case 'SET_ERROR':
      return { ...state, error: action.payload, loading: false };
    
    default:
      return state;
  }
}

// ===== Context创建 =====
interface DreamContextValue {
  state: AppState;
  dispatch: React.Dispatch<AppAction>;
}

const DreamContext = createContext<DreamContextValue | undefined>(undefined);

// ===== Provider =====
export function DreamProvider({ children }: { children: ReactNode }) {
  const [state, dispatch] = useReducer(dreamReducer, initialState);

  // 初始化：加载用户ID和梦境数据
  useEffect(() => {
    try {
      const userId = getUserId();
      const dreams = loadDreams();
      dispatch({ type: 'SET_DREAMS', payload: dreams });
    } catch (err) {
      dispatch({ type: 'SET_ERROR', payload: '加载数据失败，请刷新页面重试' });
    }
  }, []);

  return (
    <DreamContext.Provider value={{ state, dispatch }}>
      {children}
    </DreamContext.Provider>
  );
}

// ===== Consumer Hook =====
export function useDreamContext(): DreamContextValue {
  const context = useContext(DreamContext);
  if (!context) {
    throw new Error('useDreamContext must be used within DreamProvider');
  }
  return context;
}
```


### 1.3 数据同步流程

```
1. App初始化 → DreamProvider挂载
2. useEffect触发 → getUserId() → loadDreams()
3. dispatch SET_DREAMS → 所有页面通过useDreamContext()获取最新数据
4. 用户创建/编辑/删除梦境 → dispatch对应action → 同时调用storage.ts写入localStorage
5. 数据始终保持 Context State ↔ localStorage 双写同步
```

## 二、自定义Hooks

### 2.1 useDreams

```typescript
// src/hooks/useDreams.ts
// 对storage.ts操作的便捷封装，自动同步Context

export function useDreams() {
  const { state, dispatch } = useDreamContext();

  const addDream = (input: DreamInput): Dream => {
    const dream = createDream(input);
    dispatch({ type: 'ADD_DREAM', payload: dream });
    return dream;
  };

  const editDream = (id: string, input: DreamInput): Dream => {
    const dream = updateDream(id, input);
    dispatch({ type: 'UPDATE_DREAM', payload: dream });
    return dream;
  };

  const removeDream = (id: string): void => {
    deleteDream(id);
    dispatch({ type: 'DELETE_DREAM', payload: id });
  };

  return {
    dreams: state.dreams,
    loading: state.loading,
    error: state.error,
    addDream,
    editDream,
    removeDream,
  };
}
```

### 2.2 useSpeech

```typescript
// src/hooks/useSpeech.ts
// 封装Web Speech API语音识别

export function useSpeech() {
  const [isListening, setIsListening] = useState(false);
  const [transcript, setTranscript] = useState('');
  const recognitionRef = useRef<SpeechRecognition | null>(null);

  const isSupported = typeof window !== 'undefined' &&
    ('SpeechRecognition' in window || 'webkitSpeechRecognition' in window);

  const startListening = useCallback(() => {
    if (!isSupported) return;

    const SpeechRecognition = (window as any).SpeechRecognition || (window as any).webkitSpeechRecognition;
    const recognition = new SpeechRecognition();
    recognition.lang = 'zh-CN';
    recognition.interimResults = true;
    recognition.continuous = true;

    recognition.onresult = (event: SpeechRecognitionEvent) => {
      const result = Array.from(event.results)
        .map(r => r[0].transcript)
        .join('');
      setTranscript(result);
    };

    recognition.onerror = () => {
      setIsListening(false);
    };

    recognition.onend = () => {
      setIsListening(false);
    };

    recognitionRef.current = recognition;
    recognition.start();
    setIsListening(true);
  }, [isSupported]);

  const stopListening = useCallback(() => {
    recognitionRef.current?.stop();
    setIsListening(false);
  }, []);

  // 清理
  useEffect(() => {
    return () => { recognitionRef.current?.stop(); };
  }, []);

  return { isListening, transcript, startListening, stopListening, isSupported, resetTranscript: () => setTranscript('') };
}
```

```typescript
// src/hooks/useEmotionTheme.ts
// 将情绪颜色注入CSS变量

export function useEmotionTheme(emotion: EmotionType) {
  useEffect(() => {
    const meta = EMOTION_META[emotion];
    const root = document.documentElement;
    root.style.setProperty('--emotion-color', meta.color);
    root.style.setProperty('--emotion-bg', meta.bgGradient);
    
    return () => {
      root.style.removeProperty('--emotion-color');
      root.style.removeProperty('--emotion-bg');
    };
  }, [emotion]);
}
```

## 四、所有边界状态清单

> 以下每个边界状态都必须在代码中实现对应的处理逻辑。

### 4.1 空状态

| 状态 | 触发条件 | UI表现 |
|------|----------|--------|
| 无梦境 | `dreams.length === 0` | EmptyState组件，消息"还没有梦被留下"，按钮"记录你的第一个梦" |
| 找不到梦境 | `dream`不存在（详情页/编辑页） | EmptyState组件，消息"找不到这个梦"，按钮"返回首页" |
| 某天无梦境 | 日历选中日期但无记录 | 文本提示"这一天没有梦" |
| 无搜索结果 | 搜索无匹配 | EmptyState组件，消息"没有找到匹配的梦境"（P2） |

### 4.2 输入边界

| 状态 | 触发条件 | UI表现 |
|------|----------|--------|
| 空输入提交 | `content.trim() === ''` | Toast"梦还空着呢，写点什么吧"，阻止提交 |
| 超长输入 | `content.length > 4500` | 字数统计区域显示温和提示"快要写满了呢" |
| 达到上限 | `content.length === 5000` | Textarea maxLength属性阻止继续输入 |
| 特殊字符 | HTML标签输入 | 纯文本存储，自然展示，无XSS风险 |

### 4.3 异常状态

| 状态 | 触发条件 | UI表现 |
|------|----------|--------|
| localStorage不可用 | 隐私模式/存储满 | 页面顶部显示警告条"浏览器的存储功能不可用，梦境数据可能无法保存" |
| 语音不支持 | 浏览器不支持Web Speech API | VoiceInput组件返回null，不渲染语音按钮 |
| 语音识别错误 | Web Speech API报错 | 静默停止录音，Toast提示"语音识别出错，请使用文字输入" |
| 路由参数无效 | `/dream/invalidId` | EmptyState"找不到这个梦" |

### 4.4 操作确认

| 操作 | 确认方式 | 文案 |
|------|----------|------|
| 删除梦境 | ConfirmDialog二次确认 | 标题"删除梦境"，消息"删除后，这个梦就真的消失了"，确认按钮"确认删除" |
| 放弃编辑 | 无（编辑页直接返回，不提示） | — |

## 五、交互细节规范

### 5.1 页面过渡

```
所有页面切换使用CSS transition：
- opacity: 0 → 1 (300ms ease)
- transform: translateY(8px) → translateY(0) (300ms ease)
```

### 5.2 按钮状态

```
所有操作按钮必须有三种状态：
1. default: 正常可点击
2. disabled: 灰色半透明，cursor: not-allowed
3. hover: 轻微颜色加深或上浮

Loading状态：
- 提交按钮：disabled + 文案变更（"正在安放这个梦..."）
- 列表页面：首次加载显示"加载中..."（仅loading=true时）
```

### 5.3 Toast规范

```
- 位置：页面底部居中，距底部24px
- 持续时间：2秒自动消失
- 成功：绿色背景 + 白色文字
- 错误：红色背景 + 白色文字
- 信息：深色背景 + 白色文字
- 动画：从底部滑入 → 停留 → 滑出
```

## 六、存储同步校验

```typescript
// 每次操作后的验证逻辑（在storage.ts中实现）

function saveDreams(dreams: Dream[]): void {
  try {
    const userId = getUserId();
    const key = `${STORAGE_PREFIX}_${userId}_dreams`;
    localStorage.setItem(key, JSON.stringify(dreams));
  } catch (error) {
    console.error('Failed to save dreams to localStorage:', error);
    throw new Error('梦境保存失败，请检查浏览器存储空间');
  }
}
```
